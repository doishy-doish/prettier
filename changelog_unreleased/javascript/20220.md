#### Keep Flow return type comment after `)` instead of moving it into the parameters (#20220 by @doishy-doish)

<!-- prettier-ignore -->
```jsx
// Input
async function doStuff(
    arg1 /*: string */, // The first arg
    arg2 /*: string */ // The second arg
) /*: Promise<string> */ // The return value
{
  console.log('thanks!');
}

// Prettier stable
async function doStuff(
  arg1 /*: string */, // The first arg
  arg2 /*: string */, // The second arg
  /*: Promise<string> */
) {
  // The return value
  console.log("thanks!");
}

// Prettier main
async function doStuff(
  arg1 /*: string */, // The first arg
  arg2 /*: string */, // The second arg
) /*: Promise<string> */ {
  // The return value
  console.log("thanks!");
}
```
